\begin{frame}{Main title}{Subtitle}
	This is how you create titles and subtitles for your slides:
	\begin{codesnippet}
		\command{begin}\{frame\}\{Main title\}\{Subtitle\} \\
		\ind ... content ... \\
		\command{end}\{frame\}
	\end{codesnippet} \\
	\vspace{0.5cm}
	You can leave out the titles and subtitles by leaving them blank:
	\begin{codesnippet}
		\command{begin}\{frame\}\{\}\{\} \\
		\ind ... content ... \\
		\command{end}\{frame\}
	\end{codesnippet} \\
	\vspace{0.5cm}
	Or by leaving them out entirely:
	\begin{codesnippet}
		\command{begin}\{frame\} \\
		\ind ... content ... \\
		\command{end}\{frame\}
	\end{codesnippet}
\end{frame}

\begin{frame}{Bulleted lists}{Using the {\tt itemize} environment}
	This is how you make bullet points:
	\begin{itemize}
		\item Bullet point 1
		\item Bullet point 2
		\item Bullet point 3
	\end{itemize}

	Use the {\tt itemize} environment and {\tt $\backslash$item} for each bullet:\\
	\vspace{0.2cm}
	\begin{codesnippet}
		\command{begin}\{itemize\} \\
		\ind \ind \command{item} Bullet point 1 \\
		\ind \ind \command{item} Bullet point 2 \\
		\ind \ind \command{item} Bullet point 3 \\
		\command{end}\{itemize\}
	\end{codesnippet}
\end{frame}

\begin{frame}{Bulleted lists 2}{Using the {\tt outline} environment}
	Normally, to make levels of bullet points you need to nest {\tt itemize} environments. \\

	It's easier to use {\tt outline} environments instead. For this, you need to load the {\tt outlines} package in the preamble, and use {\tt $\backslash$1} for level 1 , {\tt $\backslash$2} for level 2 etc. \\
	\vspace{0.2cm}
	\begin{codesnippet}
		\command{usepackage}\{outlines\} \\
		... \\
		\command{begin}\{outline\} \\
		\ind \ind \command{1} Bullet point 1 \\
		\ind \ind \command{2} Bullet point 2 \\
		\ind \ind \command{3} Bullet point 3 \\
		\command{end}\{outline\}
	\end{codesnippet}
\end{frame}

\begin{frame}{Bulleted lists 3}{Nested lists with the {\tt itemize} environment}
	A nested list will change bullet shapes and font sizes automatically:
	\begin{itemize}
		\item Bullet point 1
		      \begin{itemize}
			      \item Bullet point 2
			            \begin{itemize}
				            \item Bullet point 3
			            \end{itemize}
		      \end{itemize}
	\end{itemize}
\end{frame}

\begin{frame}{Bulleted lists 4}{Nested lists with the {\tt outline} environment}
	A nested list will change bullet shapes and font sizes automatically:
	\begin{outline}
		\1 Bullet point 1
		\2 Bullet point 2
		\3 Bullet point 3
	\end{outline}
\end{frame}

\begin{frame}{Numbered lists}{Using the {\tt enumerate} environment}
	The {\tt enumerate} environment works similarly to {\tt itemize}, but creates numbered lists instead of bulleted lists:
	\begin{enumerate}
		\item Bullet point 1
		\item Bullet point 2
		\item Bullet point 3
	\end{enumerate}
	You also need to nest these environments for multiple levels. \\
	For more customisation options with lists using the aforementioned environments/packages, you can refer to their \LaTeX{} documentation on CTAN.
\end{frame}

		\command{begin}\{frame\}[plain] \\
		\ind ... content ... \\

	\end{codesnippet}

\begin{frame}{Slide with no footer}
	If you use {\tt \textbackslash hidefooter} before your frame, \\
	you can remove footers locally on a slide (remember to use {\tt \textbackslash showfooter} afterwards): \\
	\begin{codesnippet}
		\command{hidefooter} \\
		\command{begin}\{frame\}\{Title\} \\
		\ind ... content ... \\
		\command{end}\{frame\} \\
		\command{showfooter}
	\end{codesnippet} \\
	\vspace{0.5cm}
	There are three options:
	\begin{itemize}
		\item {\tt \textbackslash hidefooter} to hide the entire footer
		\item {\tt \textbackslash hidefooterleft} to hide only short field names on the bottom left
		\item {\tt \textbackslash hideframenumber} to hide only the slide numbering on the bottom right
	\end{itemize}
	These options can also be set globally in the preamble.
\end{frame}
